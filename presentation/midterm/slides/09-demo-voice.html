async def classify_intent(text: str, robot: str) -> dict:
    system = prompts.classify_system(robot)
    messages = [
        {"role": "system", "content": system},
        {"role": "user", "content": text},
    ]
    raw = await _ollama_chat(
        messages=messages, num_predict=60, temperature=0.1,
    )
    return json.loads(raw)   # {"mode": "숨바꼭질", "params": {...}}
