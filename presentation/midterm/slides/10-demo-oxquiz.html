policy_rule.py
class OXRulePolicy(Policy):
    def decide(self, obs: dict) -> str:
        answer = obs.get("answer")
        if answer == "O":
            return "trajectory:o_motion"
        elif answer == "X":
            return "trajectory:x_motion"
        return "wait"
